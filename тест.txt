Вот готовый код для ArticleParamsForm.tsx. Просто скопируй его (или адаптируй под свои названия классов):

import { Button } from 'src/ui/button';
import styles from './ArticleParamsForm.module.scss';

// Описываем, что ждём от родителя
interface Props {
  bgColor: string;
  setBgColor: (val: string) => void;
  fontSize: string | number;
  setFontSize: (val: string | number) => void;
  onApply: () => void;
  onReset: () => void;
}

export const ArticleParamsForm = ({
  bgColor,
  setBgColor,
  fontSize,
  setFontSize,
  onApply,
  onReset,
}: Props) => {
  return (
    <form className={styles.form}>
      {/* Поле цвета */}
      <div className={styles.field}>
        <label>Цвет фона:</label>
        <input
          type="color"
          value={bgColor}
          onChange={(e) => setBgColor(e.target.value)}
        />
      </div>

      {/* Поле размера */}
      <div className={styles.field}>
        <label>Размер шрифта:</label>
        <input
          type="number"
          value={fontSize}
          onChange={(e) => setFontSize(e.target.value)}
          min="10"
          max="100"
        />
      </div>

      {/* Кнопки */}
      <div className={styles.bottomContainer}>
        {/* Кнопка "Сбросить": просто вызываем функцию из родителя */}
        <Button
          title="Сбросить"
          type="clear"
          onClick={onReset}
        />

        {/* Кнопка "Применить": вызываем функцию применения */}
        <Button
          title="Применить"
          type="apply"
          onClick={onApply}
        />
      </div>
    </form>
  );
};
Важно: Я убрал htmlType у кнопок. Теперь они просто кнопки, которые вызывают JS-функции. Это даёт нам полный контроль.
